\section{Chapter Compendium}

\begin{compendium}[Linear Transformations]
    \textbf{Definition}: $T: U \to V$ linear iff $T(au + bv) = aT(u) + bT(v)$

    \textbf{Category $K$-Vect}: $\Hom[K]{U}{V} = \Set{T : U \to V \mid T \text{ linear}}$

    \textbf{Composition}: $T, S$ linear $\Rightarrow$ $\Composition{S}{T}$ linear (associative, identity morphisms)

    \textbf{Basic Properties}: $T(0) = 0$, $T(-v) = -T(v)$, $T(\sum a_i u_i) = \sum a_i T(u_i)$
\end{compendium}

\begin{compendium}[Universal Property (Free Objects)]
    For basis $B$ of $V$, any $f: B \to W$ extends uniquely to linear $T: V \to W$

    \textbf{Equivalent} (no isomorphism): $V$ has basis $\Leftrightarrow$ universal property $\Leftrightarrow$ $V = \bigoplus \Span{v_i}$

    \textbf{Corollary}: Every vector space is free (has basis by Zorn's lemma)

    \textbf{Corollary}: Linear maps determined by values on basis
\end{compendium}

\begin{compendium}[Subspaces Under Linear Maps]
    \textbf{Images and Preimages}: If $W \Subspace[K] U$, then
    $\DirectImage{T}{W} \Subspace[K] V$. If $W \Subspace[K] V$, then
    $\InverseImage{T}{W} \Subspace[K] U$.

    \textbf{Image-Preimage Formulas}:
    \begin{itemize}
        \item $\InverseImage{T}{\DirectImage{T}{W}} = W + \Ker{T}$ for $W \Subspace[K] U$
        \item $\DirectImage{T}{\InverseImage{T}{W'}} = W' \cap \Rng{T}$ for $W' \Subspace[K] V$
    \end{itemize}

    \textbf{Span and Image Commute}: $\Span[K]{\DirectImage{T}{S}} = \DirectImage{T}{\Span[K]{S}}$
\end{compendium}

\begin{compendium}[Fundamental Subspaces]
    \textbf{Kernel and Range}:
    \begin{itemize}
        \item $\Ker{T} = \InverseImage{T}{\{0\}} = \Set{u \in U \mid T(u) = 0} \Subspace[K] U$
        \item $\Rng{T} = \DirectImage{T}{U} = \Set{T(u) \mid u \in U} \Subspace[K] V$
    \end{itemize}

    \textbf{Dimensions}: $\Rank{T} = \Dim[K]{\Rng{T}}$, \;
    $\Nullity{T} = \Dim[K]{\Ker{T}}$
\end{compendium}

\begin{compendium}[Hom Spaces and Endomorphisms]
    \textbf{Vector Space}: $\Hom[K]{U}{V}$ is a $K$-vector space under pointwise operations

    \textbf{Natural Isomorphism}: $\Hom[K]{K}{V} \cong V$ via $T \mapsto T(1_K)$

    \textbf{Endomorphism Space}: $\End[K]{V} = \Hom[K]{V}{V}$ with multiplication $S \cdot T = \Composition{S}{T}$

    \textbf{$K$-Algebra}: $(\End[K]{V}, +, \cdot)$ is a unital associative $K$-algebra;
    non-commutative when $\Dim[K]{V} \geq 2$
\end{compendium}

\begin{compendium}[Injectivity Characterizations]
    For $T: U \to V$ linear, TFAE (one chain):
    \begin{enumerate}
        \item $\Ker{T} = \{0\}$
        \item $T$ injective
        \item $\InverseImage{T}{\DirectImage{T}{W}} = W$ for every $W \Subspace[K] U$
        \item $W \mapsto \DirectImage{T}{W}$ injective on subspace lattice of $U$
        \item $\Dim[K]{\DirectImage{T}{W}} = \Dim[K]{W}$ for every $W \Subspace[K] U$
        \item lin.\ indep.\ $\to$ lin.\ indep.\
        \item every basis $\to$ lin.\ indep.\
        \item some basis $\to$ lin.\ indep.\
        \item split mono ($\Exists{S}{\Composition{S}{T} = \Identity{U}}$)
        \item monic (left-cancellable)
    \end{enumerate}
\end{compendium}

\begin{compendium}[Surjectivity Characterizations]
    For $T: U \to V$ linear, TFAE (one chain):
    \begin{enumerate}
        \item $\Rng{T} = V$
        \item $T$ surjective
        \item $\DirectImage{T}{\InverseImage{T}{W}} = W$ for every $W \Subspace[K] V$
        \item $W \mapsto \InverseImage{T}{W}$ injective on subspace lattice of $V$
        \item $\Codim[K]{\InverseImage{T}{W}}{U} = \Codim[K]{W}{V}$
              for every $W \Subspace[K] V$
        \item gen.\ set $\to$ gen.\ set
        \item every basis $\to$ gen.\ set
        \item some basis $\to$ gen.\ set
        \item split epi ($\Exists{S}{\Composition{T}{S} = \Identity{V}}$)
        \item epic (right-cancellable)
    \end{enumerate}
\end{compendium}

\begin{compendium}[Bijectivity and Isomorphism]
    \textbf{Bijectivity TFAE} (one chain): bijective $\Rightarrow$
    preimage/image formulas $\Rightarrow$ lattice isomorphism $\Rightarrow$
    dim + codim preserved $\Rightarrow$ every/some basis $\to$ basis $\Rightarrow$
    isomorphism $\Rightarrow$ monic + epic $\Rightarrow$ bijective

    \textbf{Isomorphism}: $U \cong V \Leftrightarrow \Dim[K]{U} = \Dim[K]{V}$

    \textbf{Coordinate Isomorphism}: $\phi_{\mathcal{B}}: V \xrightarrow{\cong} K^{(I)}$
    (depends on ordered basis $\mathcal{B}$)
\end{compendium}

\begin{compendium}[Rank-Nullity Theorem]
    \textbf{Theorem}: $\Dim[K]{U} = \Rank{T} + \Nullity{T}$

    \textbf{Proof idea}: Extend $\Ker{T}$ basis to $U$ basis; remaining vectors map to
    $\Rng{T}$ basis

    \textbf{Equal Dimension Equivalence}: For $T: U \to V$ with
    $\Dim[K]{U} = \Dim[K]{V}$ finite: injective $\Leftrightarrow$ surjective
    $\Leftrightarrow$ bijective

    \textbf{Endomorphism Corollary}: For $T: V \to V$ with $V$ finite-dimensional:
    injective $\Leftrightarrow$ surjective $\Leftrightarrow$ bijective
\end{compendium}
