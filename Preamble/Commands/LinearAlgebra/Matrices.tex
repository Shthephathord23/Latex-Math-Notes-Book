% Preamble/Commands/LinearAlgebra/Matrices.tex
% Convention: for f: K^X -> K^Y, the matrix lies in Mat_{Y,X}(K)
% Usage: \MatSpace{Y}{X}[K] → Mat_{Y,X}(K)  (#1 = rows/target, #2 = cols/source)
% Usage: \CFMat{Y}{X}[K]   → Mat_{Y,(X)}(K)  (column-finite)

\NewDocumentCommand{\MatSpace}{m m o}
{
    \IfValueTF{#3}
    {
        \operatorname{Mat}_{#1, #2} \left( #3 \right)
    }
    {
        \operatorname{Mat}_{#1, #2}
    }
}

% Column-finite matrix space: \CFMat{Y}{X}[K] → Mat_{Y,(X)}(K)
\NewDocumentCommand{\CFMat}{m m o}
{
    \IfValueTF{#3}
    {
        \operatorname{Mat}_{(#1), #2} \left( #3 \right)
    }
    {
        \operatorname{Mat}_{(#1), #2}
    }
}

% Override physics package \Transpose
\RenewDocumentCommand{\Transpose}{tp m}
{
    \IfBooleanTF{#1}
    {
        {\left( #2 \right)}^\mathsf{T}
    }
    {
        {#2}^\mathsf{T}
    }
}

\NewDocumentCommand{\Adjoint}{tp m}
{
    \IfBooleanTF{#1}
    {
        {\left( #2 \right)}^*
    }
    {
        {#2}^*
    }
}

% Override physics package \Det
\RenewDocumentCommand{\Det}{m}
{
    \det \left( #1 \right)
}

% Override physics package \Trace
\RenewDocumentCommand{\Trace}{m}
{
    \operatorname{Tr} \left( #1 \right)
}

\NewDocumentCommand{\Adjugate}{m}
{
    \operatorname{adj} \left( #1 \right)
}

\NewDocumentCommand{\Hermitian}{tp m}
{
    \IfBooleanTF{#1}
    {
        {\left( #2 \right)}^\dagger
    }
    {
        {#2}^\dagger
    }
}

\NewDocumentCommand{\CoordVec}{o m}
{
    \IfValueTF{#1}
    {
        \left[#2\right]^{#1}
    }
    {
        \left[#2\right]
    }
}

\NewDocumentCommand{\CoordMatrix}{o o m}
{
    \IfValueTF{#2}
    {
        \left[#3\right]_{#1}^{#2}
    }
    {
        \IfValueTF{#1}
        {
            \left[#3\right]_{#1}
        }
        {
            \left[#3\right]
        }
    }
}


% Identity matrix: \IdentityMat{\kappa} → I_{\kappa}
\NewDocumentCommand{\IdentityMat}{m}
{
    I_{#1}
}

% Equivalence of linear maps / matrices:  T = q o S o p^{-1}   (A = Q B P^{-1})
% Usage: $S \Equivalent T$,  $A \Equivalent B$
\NewDocumentCommand{\Equivalent}{}
{
    \approx
}

% Similarity (conjugacy) of endomorphisms / square matrices:  T = q^{-1} o S o q   (A = Q^{-1} B Q)
% Usage: $S \Similar T$,  $A \Similar B$
\NewDocumentCommand{\Similar}{}
{
    \sim
}

% Domain (column) equivalence: only the source basis changes.  T = S o p^{-1}   (A = B P^{-1})
% Right multiplication by an invertible factor; remixes the columns (indexed by the source).
% Usage: $S \DomainEquivalent T$,  $A \DomainEquivalent B$
\NewDocumentCommand{\DomainEquivalent}{}
{
    \mathrel{\overset{\mathrm{col}}{\approx}}
}

% Codomain (row) equivalence: only the target basis changes.  T = q o S   (A = Q B)
% Left multiplication by an invertible factor; remixes the rows (indexed by the target).
% Usage: $S \CodomainEquivalent T$,  $A \CodomainEquivalent B$
\NewDocumentCommand{\CodomainEquivalent}{}
{
    \mathrel{\overset{\mathrm{row}}{\approx}}
}
